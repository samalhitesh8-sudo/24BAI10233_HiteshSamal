SQL Questions
1.
List the Employee Name, their Department Name, and the Location Name for all employees.

SELECT emp.employee_name, dept.dept_name, dept.location_id
FROM Employees emp
JOIN Departments dept ON emp.dept_id = dept.dept_id;

2.
Retrieve the Project Name, Client Name, and the Total Hours Worked on that project, but only for projects managed by the 'Consulting' department.

SELECT proj.project_name, proj.client_name, SUM(asgn.hours_worked) AS total_hours
FROM Projects proj
JOIN Departments dept ON proj.dept_id = dept.dept_id
JOIN Assignments asgn ON proj.project_id = asgn.project_id
WHERE dept.dept_name = 'Consulting'
GROUP BY proj.project_id, proj.project_name, proj.client_name;

3.
Find the names of Employees who have worked on a project with Client Name 'GlobalTech Solutions'. List the Employee Name and Project Name.

SELECT DISTINCT emp.employee_name, proj.project_name
FROM Employees emp
JOIN Assignments asgn ON emp.employee_id = asgn.employee_id
JOIN Projects proj ON asgn.project_id = proj.project_id
WHERE proj.client_name = 'GlobalTech Solutions';

4.
List all Departments that have no Employees currently assigned to them.

SELECT dept.dept_name
FROM Departments dept
LEFT JOIN Employees emp ON dept.dept_id = emp.dept_id
WHERE emp.employee_id IS NULL;

5.
Show the Manager Name, the Department Name they manage, and the total number of direct reports they have.

SELECT mgr.employee_name AS manager_name,dept.dept_name,
COUNT(emp.employee_id) AS direct_reports
FROM Employees emp
JOIN Employees mgr ON emp.manager_id = mgr.employee_id
JOIN Departments dept ON mgr.dept_id = dept.dept_id
GROUP BY mgr.employee_id, mgr.employee_name, dept.dept_name;

6.
Calculate the Average Hours Worked per Employee across all projects. List the Employee Name and the average hours.

SELECT emp.employee_name, AVG(asgn.hours_worked) AS avg_hours
FROM Employees emp
JOIN Assignments asgn ON emp.employee_id = asgn.employee_id
GROUP BY emp.employee_id, emp.employee_name;

8.
Identify the Client Name(s) whose projects have a total combined hours worked exceeding 500 hours.

SELECT proj.client_name, SUM(asgn.hours_worked) AS total_hours
FROM Projects proj
JOIN Assignments asgn ON proj.project_id = asgn.project_id
GROUP BY proj.client_name
HAVING SUM(asgn.hours_worked) > 500;

10.
Calculate the total number of employees assigned to each Project, but only include projects that have more than 5 employees assigned.

SELECT proj.project_name, COUNT(DISTINCT asgn.employee_id) AS emp_count
FROM Projects proj
JOIN Assignments asgn ON proj.project_id = asgn.project_id
GROUP BY proj.project_id, proj.project_name
HAVING COUNT(DISTINCT asgn.employee_id) > 5;

12.
Retrieve the Department Name(s) that have employees assigned to more than one project.

SELECT DISTINCT dept.dept_name
FROM Departments dept
JOIN Employees emp ON dept.dept_id = emp.dept_id
JOIN Assignments asgn ON emp.employee_id = asgn.employee_id
GROUP BY dept.dept_id, dept.dept_name, emp.employee_id
HAVING COUNT(DISTINCT asgn.project_id) > 1;

13.
Write a query to list the Employee Name and Project Name for any employee who has worked less than 10 hours on a project.

SELECT emp.employee_name, proj.project_name, asgn.hours_worked
FROM Employees emp
JOIN Assignments asgn ON emp.employee_id = asgn.employee_id
JOIN Projects proj ON asgn.project_id = proj.project_id
WHERE asgn.hours_worked < 10;

14.
Find the Project Name(s) that have no assigned employees.

SELECT proj.project_name
FROM Projects proj
LEFT JOIN Assignments asgn ON proj.project_id = asgn.project_id
WHERE asgn.assignment_id IS NULL;

16.
Assuming the Assignments table has a start_date column, find the Employee Name and their oldest project start date.

SELECT emp.employee_name, MIN(asgn.start_date) AS oldest_start_date
FROM Employees emp
JOIN Assignments asgn ON emp.employee_id = asgn.employee_id
GROUP BY emp.employee_id, emp.employee_name;

17.
Write a MySQL query using the DATE_FORMAT function to list all projects that were active in the year 2023.

SELECT project_name, start_date FROM Projects
WHERE DATE_FORMAT(start_date, '%Y') = '2023';

18.
Use the MySQL-specific IFNULL function to list all Employee Names. If an employee has no manager, display "CEO" instead of NULL.

SELECT employee_name, IFNULL(CAST(manager_id AS CHAR), 'CEO') AS manager
FROM Employees;

19.
Find the top 3 Departments based on the total hours worked by all their employees across all projects.

SELECT dept.dept_name, SUM(asgn.hours_worked) AS total_hours
FROM Departments dept
JOIN Employees emp ON dept.dept_id = emp.dept_id
JOIN Assignments asgn ON emp.employee_id = asgn.employee_id
GROUP BY dept.dept_id, dept.dept_name
ORDER BY total_hours DESC
LIMIT 3;

20.
Display the Project Name and a status ("High Activity" if total hours > 300, "Normal" otherwise) using the CASE statement.

SELECT proj.project_name, SUM(asgn.hours_worked) AS total_hours,
CASE
   WHEN SUM(asgn.hours_worked) > 300 THEN 'High Activity'
   ELSE 'Normal'
END AS status
FROM Projects proj
LEFT JOIN Assignments asgn ON proj.project_id = asgn.project_id
GROUP BY proj.project_id, proj.project_name;
